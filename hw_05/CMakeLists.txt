# ============================================================================
# Корневой CMakeLists.txt (Задания 5.I - 5.V)
# ============================================================================

# Указываем минимально необходимую версию CMake
cmake_minimum_required(VERSION 3.14)

# Задаем имя проекта и язык программирования C++
project(HypoProject CXX)

# Устанавливаем современный стандарт языка C++17
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# ----------------------------------------------------------------------------
# Задание 5.IV: Опция для включения или выключения сборки тестов
# По умолчанию включена (ON).
# Чтобы собрать проект БЕЗ тестов, передайте флаг:
#   cmake -B build -DBUILD_TESTS=OFF
# ----------------------------------------------------------------------------
option(BUILD_TESTS "Включить сборку модульных тестов Google Test" ON)

# ----------------------------------------------------------------------------
# Задание 5.II: Подключение библиотеки из поддиректории hypo_lib
# CMake переходит в папку hypo_lib и читает находящийся там CMakeLists.txt
# ----------------------------------------------------------------------------
add_subdirectory(hypo_lib)

# ----------------------------------------------------------------------------
# Создание основного исполняемого файла приложения main_app
# ----------------------------------------------------------------------------
add_executable(main_app main.cpp)

# Связываем наше приложение с библиотекой hypo_lib
# Благодаря этому main.cpp видит заголовочные файлы библиотеки hypo.hpp и hypo_mod.hpp
target_link_libraries(main_app PRIVATE hypo_lib)

# ----------------------------------------------------------------------------
# Задание 5.III и 5.IV: Подключение каталога с тестами (если опция включена)
# ----------------------------------------------------------------------------
if(BUILD_TESTS)
    message(STATUS ">>> Сборка тестов Google Test ВКЛЮЧЕНА (BUILD_TESTS = ON)")
    enable_testing()
    add_subdirectory(tests)
else()
    message(STATUS ">>> Сборка тестов Google Test ВЫКЛЮЧЕНА (BUILD_TESTS = OFF)")
endif()
