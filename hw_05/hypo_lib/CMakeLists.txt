# ============================================================================
# CMakeLists.txt для библиотеки hypo_lib (Задания 5.II и 5.V)
# ============================================================================
# В задании 5.V функции библиотеки стали шаблонными (template).
# Код шаблонных функций должен находиться целиком в заголовочных файлах (.hpp),
# поэтому у библиотеки нет отдельных .cpp файлов с реализацией.
# В CMake такие библиотеки называются "header-only" и создаются
# с ключевым словом INTERFACE (add_library(... INTERFACE)).
# ============================================================================

add_library(hypo_lib INTERFACE)

# target_include_directories с модификатором INTERFACE указывает компилятору,
# где искать заголовочные файлы библиотеки (hypo.hpp и hypo_mod.hpp).
# Все цели (наш main_app и тесты), которые связываются с hypo_lib,
# автоматически получают доступ к этой директории для директивы #include.
target_include_directories(hypo_lib INTERFACE
    ${CMAKE_CURRENT_SOURCE_DIR}
)
